export async function triageClaim(claim: Claim) {
  const evidence = await retrieve(claim, { topK: 12 });
  const decision = await agent.run({ claim, evidence });

  // Confidence below threshold? A human decides.
  if (decision.confidence < 0.97) {
    return escalate(decision, "needs-review");
  }
  return resolve(decision, { audit: true });
}

Process

Four phases. No surprises.

A fixed operating rhythm, a demo every Friday, and a written decision log from kickoff to handover.

01

Diagnose

Two weeks embedded with your team. We map the domain, the data, and the risk — and tell you honestly if you don't need us.

02

Design

Architecture and interface designed together, prototyped in code. You review working software, not slideware.

03

Build

Senior engineers only — no bait-and-switch staffing. Weekly demos, trunk-based delivery, performance budgets in CI.

04

Run & hand over

We operate what we ship, then transfer it with docs, runbooks, and pairing until your team is faster in the codebase than we are.
